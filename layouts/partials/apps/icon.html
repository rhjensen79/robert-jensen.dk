{{/* Self-hosted, resized WebP app icon. Args: app (map), size (px, rendered), alt (optional), decorative (bool). Fetches the 256px PNG from 12f.dk and serves a 2x WebP from this site; hotlinks the PNG if the fetch fails. decorative=true renders a background-image span instead of an , so theme's medium-zoom (binds every img in .post__content) leaves it alone. */}} {{- $size := .size | default 48 -}} {{- $src := printf "https://www.12f.dk/images/apps/%s.png" .app.slug -}} {{- $out := $src -}} {{- with try (resources.GetRemote $src) -}} {{- with .Value -}} {{- $px := mul $size 2 -}} {{- $img := .Resize (printf "%dx%d webp q85" $px $px) -}} {{- $out = ($img | resources.Copy (printf "images/apps/%s-%d.webp" $.app.slug $px)).RelPermalink -}} {{- end -}} {{- end -}} {{- if .decorative -}} {{- else -}} {{- end -}}
